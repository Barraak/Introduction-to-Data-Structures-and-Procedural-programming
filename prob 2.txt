PROCEDURE dot_product(v1, v2 : ARRAY_OF FLOAT, VAR ps : FLOAT)
VAR
    i : INTEGER;
BEGIN
    ps := 0;
    FOR i FROM 0 TO v1.length - 1 DO
        ps := ps + (v1[i] * v2[i]);
    END_FOR
END_PROCEDURE


ALGORITHM Orthogonal_Vectors_Procedure
VAR
    v1, v2 : ARRAY_OF FLOAT[3];
    ps : FLOAT;
    i, k, n : INTEGER;
BEGIN
    Write("Entrez le nombre de paires de vecteurs: ");
    Read(n);

    FOR k FROM 1 TO n DO
        Write("Saisie des éléments du vecteur v1 :");
        FOR i FROM 0 TO v1.length - 1 DO
            Read(v1[i]);
        END_FOR

        Write("Saisie des éléments du vecteur v2 :");
        FOR i FROM 0 TO v2.length - 1 DO
            Read(v2[i]);
        END_FOR

        dot_product(v1, v2, ps);

        IF (ps = 0) THEN
            Write("Les vecteurs de la paire ", k, " sont orthogonaux.");
        ELSE
            Write("Les vecteurs de la paire ", k, " ne sont PAS orthogonaux.");
        END_IF
    END_FOR
END
