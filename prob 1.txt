ALGORITHM Distinct_Elements_Sum
VAR
    set1 : ARRAY_OF INTEGER[4] := [3, 1, 7, 9];
    set2 : ARRAY_OF INTEGER[5] := [2, 4, 1, 9, 3];
    i, j : INTEGER;
    found : BOOLEAN;
    sum : INTEGER := 0;
BEGIN
    FOR i FROM 0 TO set1.length - 1 DO
        found := FALSE;
        FOR j FROM 0 TO set2.length - 1 DO
            IF (set1[i] = set2[j]) THEN
                found := TRUE;
                BREAK;
            END_IF
        END_FOR
        
        IF (found = FALSE) THEN
            sum := sum + set1[i];
        END_IF
    END_FOR

    FOR i FROM 0 TO set2.length - 1 DO
        found := FALSE;
        FOR j FROM 0 TO set1.length - 1 DO
            IF (set2[i] = set1[j]) THEN
                found := TRUE;
                BREAK;
            END_IF
        END_FOR
        
        IF (found = FALSE) THEN
            sum := sum + set2[i];
        END_IF
    END_FOR

    Write("Sum of distinct elements: ", sum);
END
